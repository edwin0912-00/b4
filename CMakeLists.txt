cmake_minimum_required(VERSION 3.24)
project(BeforeEffects VERSION 0.7.0 LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Widgets)
# Qt 6.9 exports a redundant AGL link flag, absent from the macOS 26 SDK.
# QtGui itself needs no AGL symbols; retain its OpenGL link and installed binary.
if(APPLE AND Qt6_VERSION VERSION_LESS 6.10)
    get_target_property(motion_gl_links WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
    list(FILTER motion_gl_links EXCLUDE REGEX "AGL")
    set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${motion_gl_links}")
endif()
enable_testing()
add_library(motion_core src/project.cpp src/evaluate.cpp src/project_io.cpp src/render.cpp src/effects.cpp src/display_transform.cpp src/editor.cpp src/render_worker.cpp src/media.mm src/audio_mix.cpp)
target_include_directories(motion_core PUBLIC src)
target_link_libraries(motion_core PUBLIC Qt6::Core Qt6::Gui "-framework AVFoundation" "-framework Foundation" "-framework CoreMedia" "-framework CoreGraphics" "-framework CoreVideo" "-framework AudioToolbox" "-framework ImageIO")
set_source_files_properties(src/media.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
target_compile_options(motion_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(motion_checks tests/checks.cpp)
target_link_libraries(motion_checks PRIVATE motion_core)
target_compile_definitions(motion_checks PRIVATE MOTION_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core COMMAND motion_checks)
set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_library(motion_ui src/main_window.cpp src/main_window_media.cpp src/timeline.cpp src/graph.cpp src/property_panel.cpp src/keyframe_dialog.cpp)
target_link_libraries(motion_ui PUBLIC motion_core Qt6::Widgets)
target_compile_options(motion_ui PRIVATE -Wall -Wextra)
add_executable(BeforeEffects MACOSX_BUNDLE src/main.cpp)
target_link_libraries(BeforeEffects PRIVATE motion_ui)
set_target_properties(BeforeEffects PROPERTIES OUTPUT_NAME "Before Effects" MACOSX_BUNDLE_GUI_IDENTIFIER "org.independent.beforeeffects" MACOSX_BUNDLE_BUNDLE_NAME "Before Effects" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
add_custom_command(TARGET BeforeEffects POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/font" "$<TARGET_BUNDLE_DIR:BeforeEffects>/Contents/Resources/font")
add_executable(motion_ui_checks tests/ui_checks.cpp)
target_link_libraries(motion_ui_checks PRIVATE motion_ui)
target_compile_definitions(motion_ui_checks PRIVATE MOTION_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME ui COMMAND motion_ui_checks)
set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)

target_compile_definitions(motion_benchmark PRIVATE MOTION_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
